\begin{frame}{Motivation}
\begin{columns}[c]\begin{column}{.55\linewidth}
\textbf{Find the primary vertices and their tracks at the HL-LHC.}
\par\vspace{1em}
\begin{itemize}\setlength\itemsep{.7em}
\item About $200$ interactions per crossing along a few cm of beam line.
\item PV-Finder: a neural network turns tracks into a vertex-density histogram.
\item A graph neural network (GNN) assigns tracks to the vertices.
\item Compared with AMVF on the same events, with the same definitions.
\end{itemize}
\end{column}\begin{column}{.42\linewidth}\centering
\bigvalue{$\langle\mu\rangle=200$}{HL-LHC, ITk}
\par\vspace{1.2em}
\begin{enumerate}\setlength\itemsep{.35em}\small
\item[\textcolor{pennred}{\textbf{I.}}] Method
\item[\textcolor{pennred}{\textbf{II.}}] Vertex finding at PU200
\item[\textcolor{pennred}{\textbf{III.}}] Track association and hard scatter
\item[\textcolor{pennred}{\textbf{IV.}}] Low pile-up and collision data
\item[\textcolor{pennred}{\textbf{V.}}] Timing and deployment
\end{enumerate}
\end{column}\end{columns}
\end{frame}

\begin{frame}{Track-to-Vertex Association with a GNN}
\begin{columns}[c]\begin{column}{.52\linewidth}
\small
\textcolor{mydarkblue}{\textbf{Track nodes:}} $d_0,z_0$, errors, $\theta,\phi,p_T$.\\[.6em]
\textcolor{myorange}{\textbf{Vertex nodes:}} PV-Finder position and peak height.\\[.6em]
\textbf{Edges:} each track to its $k=20$ nearest vertices, with
\[\frac{z_0-z_{\rm PV}}{\sqrt{\sigma_{z_0}^2+\sigma_{\rm PV}^2}},\quad \frac{d_0}{\sigma_{d_0}},\quad |z_0-z_{\rm PV}|.\]
\textbf{Output:} a score per edge. A track goes to its best vertex if the score passes $t=\gnnT$.\\[.6em]
Graph attention network, $\gnnParams$ parameters.

\begin{frame}{Summary}
\begin{itemize}\setlength\itemsep{.75em}
\item At PU200, PV-Finder finds $\effPVF\%$ of truth vertices, $+\effDiff$ points over AMVF, with fewer fakes.
\item Vertex--vertex resolution $\sigPVF$ mm against $\sigAMVF$ mm for AMVF.
\item With GNN track association: $\cleanTruthGNN\%$ clean vertices per truth vertex, against $\cleanTruthAMVF\%$ for AMVF.
\item Hard scatter: selected in $\hsSelGNN\%$ of events ($\hsSelAMVF\%$ AMVF); position resolution $\hsResGNN~\mu$m ($\hsResAMVF~\mu$m AMVF).
\item Validated on Run 2 and Run 3 collision data.
\item Running in Athena on CPU and GPU.
\end{itemize}
\par\vspace{.6em}
\textbf{Next:} freeze the HL-LHC density model and GNN, and complete the paper draft.
\end{frame}
