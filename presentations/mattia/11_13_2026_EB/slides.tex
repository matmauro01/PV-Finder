\begin{frame}{Definitions after Track Association}
\begin{columns}[t]\begin{column}{.49\linewidth}
\textbf{Vertex categories, from track content}
\par\vspace{.4em}\footnotesize
Every vertex with $\geq2$ tracks, both algorithms. Tracks are counted, not weighted. Dominant truth vertex: the one that contributes most of the vertex tracks.
\begin{itemize}\setlength\itemsep{.25em}
\item \textbf{Clean:} $\geq70\%$ of its tracks from the dominant truth vertex.
\item \textbf{Merged:} $<70\%$ from the dominant truth vertex.
\item \textbf{Split:} another vertex with the same dominant truth vertex has a larger $\sum p_T^2$.
\item \textbf{Fake:} the dominant group is tracks with no truth vertex.
\end{itemize}
\textbf{Clean / truth:} Clean vertices divided by truth vertices with $\geq2$ tracks.
\end{column}\begin{column}{.49\linewidth}
\textbf{Reference assignments to the PV-Finder vertices}
\par\vspace{.4em}\footnotesize
\begin{itemize}\setlength\itemsep{.25em}
\item \textbf{Nearest $z_0$:} each track to the vertex nearest in $z$.
\item \textbf{$z_0$ within $3\sigma$:} the same, only if $|z_0-z_{\rm PV}|<3\,\sigma(z_0)$. $\sigma(z_0)$ is the track's own $z_0$ uncertainty from the track fit.
\item \textbf{One to one (finder cap):} vertices paired with truth vertices within $\windowMM$ mm, closest pairs first, each used once; divided by truth vertices with $\geq2$ tracks.
\item \textbf{Truth-guided (simulation only):} each track to the vertex nearest its own truth vertex, if within $\windowMM$ mm. The best any assignment can do with these vertices.
\end{itemize}
%AMVFZ0
\end{column}\end{columns}
\end{frame}

\begin{frame}{Summary}
\begin{itemize}\setlength\itemsep{.65em}
\item At PU200, before track association, PV-Finder finds $\effPVF\%$ of truth vertices against $\effAMVF\%$ for AMVF ($+\effDiff$ points), with fewer fakes.
\item Vertex--vertex resolution $\sigPVF$ mm against $\sigAMVF$ mm for AMVF.
\item With GNN track association: $\cleanTruthGNN\%$ clean vertices per truth vertex, against $\cleanTruthAMVF\%$ for AMVF (vertices with $\geq2$ tracks).
%HSSUMMARY
\item Hard scatter: selected in $\hsSelGNN\%$ of events with the Athena 24.0 rule, against $\hsSelAMVFrule\%$ for AMVF with the same rule ($\hsSelAMVF\%$ with its stored primary). Position resolution $\hsResGNN~\mu$m against $\hsResAMVF~\mu$m. Clean HS vertex and HS purity are below AMVF.
\item Low pile-up: $\effMC\%$ against $\effMCAMVF\%$ efficiency in Run 2 simulation; $\svtx$ about half of AMVF's in simulation, Run 2 and Run 3 data.
\item Running in Athena on CPU and GPU.
\end{itemize}
