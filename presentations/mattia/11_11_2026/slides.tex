\begin{frame}{1. Matching window}
\req{Justify the matching window and use it consistently.}
\did{\item Fixed the window at $0.1694$~mm $=3\times$ the 68\% longitudinal position residual of AMVF ($56.45$~\textmu m). $0.1$~mm for HS studies only.
\item Made it the code default; audited every window used in the note.}
\res{\item Previous production evaluations used the fitted $\svtx$ window ($0.229$~mm), not $0.1694$~mm.}
\vspace{.3em}
{\centering\small
\begin{tabular}{lc}
\toprule
Window & PV-Finder $-$ AMVF efficiency (points) \\
\midrule
$0.10$~mm & $+3.12\pm0.06$ \\
$0.15$~mm & $+3.37$ \\
$0.1694$~mm & $+3.36\pm0.06$ \quad ($83.16\pm0.08\%$ vs $79.80\pm0.09\%$) \\
\bottomrule
\end{tabular}\par}
\takeaway{The PV-Finder margin is stable at about $+3.1$ to $+3.4$ points for any window from $0.10$ to $0.17$~mm.}
\end{frame}

%----------------------------------------------------------------------
\begin{frame}{2. Split vertices in the efficiency}
\req{Include split vertices in the reconstruction-efficiency definition.}
\did{\item Added a split-inclusive efficiency next to the existing one, in the evaluation code.
\item Proved with tests when the two definitions can differ.}
\res{\item With closest-first matching, every truth vertex next to a split vertex is already matched by the other half of the split. The split-inclusive efficiency is therefore identical.
\item PV-Finder $83.16\%$, AMVF $79.80\%$, difference $+3.36\pm0.06$ points (at $0.1$~mm: $+3.12\pm0.06$).}
\takeaway{Splits cost fakes, not efficiency. Both definitions give the same numbers; we report both.}
\end{frame}
