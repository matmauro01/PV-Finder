\begin{frame}{1. Smaller PV-Finder for CPU}
\req{Faster CPU inference without loss of performance.}
\did{\item Narrower U-Net (s5: 112 channels), same data and loss as v6, 1+1 epochs.
\item Packed inference: only real tracks, no padding.}
\why{\item INT8 quantization: $\leq1.25\times$.}
\vspace{.2em}
{\centering\scriptsize
\begin{tabular}{lrrrr}
\toprule
Model & Params & CPU ms/event & Speed-up & $\Delta$Eff (points) \\
\midrule
v6 production & 3.55M & 1601 & 1.0 & -- \\
v6, packed & 3.55M & 1159 & 1.4 & 0 \\
s5, packed & 0.67M & 250 & 6.4 & $-0.33$ / $-0.26$ \\
\bottomrule
\end{tabular}\par}
\vspace{.4em}
{\scriptsize CPU vs CPU, ONNX Runtime, 1 thread. s5: $0.8$ fewer fakes per event.\par}
\end{frame}

\begin{frame}{2. Matching window}
\begin{columns}[T]\begin{column}{.46\linewidth}
\req{Justify the window; use it everywhere.}
\did{\item $0.1694$~mm $=3\times$ the 68\% position residual ($56.45$~\textmu m); $0.1$~mm for HS.
\item Window scan on the same events.}
\why{\item $\svtx$ ($0.229$~mm) measures vertex separation, not position.
\item Wider windows add chance matches.}
\res{\item $83.16\%$ vs $79.80\%$: $+3.36\pm0.06$ points.
\item PV-Finder ahead at every window.}
\end{column}\begin{column}{.52\linewidth}\centering
\plot{window_scan_eff_diff.png}
\end{column}\end{columns}
\end{frame}

%----------------------------------------------------------------------
\begin{frame}{3. Split vertices in the efficiency}
\req{Include split vertices in the efficiency.}
\did{\item Added a split-inclusive efficiency to the evaluation.
\item Unit tests for every matching case.}
\res{\item Identical to the standard efficiency: with closest-first matching, the truth vertex next to a split is already matched.
\item PV-Finder $83.16\%$, AMVF $79.80\%$, $+3.36\pm0.06$ points.}
\end{frame}
