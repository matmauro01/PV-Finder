\begin{frame}[shrink=15]{Motivation}
    \begin{columns}[c]
        \begin{column}{0.52\linewidth}
            \textbf{Primary-vertex finding} reconstructs the proton--proton
            interaction points along the beam axis from charged-particle tracks.
            It underlies essentially every LHC measurement.

            \vspace{0.55em}
            \textbf{PV-Finder} is a neural network that maps tracks directly to a
            1-D histogram of vertex density, then extracts positions by peak finding.

            \vspace{0.55em}
            \textbf{Key advantage:} the current default, the Adaptive Multi-Vertex
            Finder (\textbf{AMVF}), is iterative, so its runtime grows with the number
            of vertices. PV-Finder evaluates a \textbf{fixed-size map}, so its cost is
            set by the histogram, not by the event: no per-vertex iteration, no
            seeding, no refit. That matters for Run 3 ($\langle\mu\rangle\!\approx\!60$)
            and for HL-LHC ($\langle\mu\rangle\!\approx\!200$).
        \end{column}
        \begin{column}{0.48\linewidth}
            \centering
            \resizebox{0.98\linewidth}{!}{%
            \begin{tikzpicture}[x=1.1cm, y=0.9cm, >=Latex, font=\small]

\begin{frame}{Peak Finding in the Histogram Output}
    \begin{columns}[c]
        \begin{column}{0.36\linewidth}
            \small
            \begin{itemize}
                \setlength\itemsep{0.4em}
                \item 12{,}000-bin histogram (0.04\,mm/bin, $\pm240$\,mm range)
                \item Scan left-to-right: accumulate contiguous above-threshold ($\tau{=}0.01$) bins
                \item Quality cuts: ${\geq}\,3$ bins wide, integral above threshold
                \item PV position $=$ \textbf{local centroid} over the maximum $\pm3$
                    bins \textcolor{black!55}{($-13\%$ on $\sigma_{\text{vtx-vtx}}$
                    against the full-region weighted mean)}
                \item \textbf{Conjoined peaks} that never dip below $\tau$ are split at
                    their shared local minimum --- with \textbf{zero noise tolerance}
            \end{itemize}
        \end{column}

\begin{frame}{Resolution Metric: \(\sigma_{\text{vtx-vtx}}\)}
    \begin{columns}[c]
        \begin{column}{0.38\linewidth}
            \small
            \begin{itemize}
                \setlength\itemsep{0.5em}
                \item Collect all \textbf{pairwise} $\Delta z = z_i - z_j$ between reconstructed PVs in an event
                \item Pairs too close to resolve are \textbf{merged} by the peak finder $\Rightarrow$ $\Delta z$ suppressed near 0
                \item Fit a \textbf{sigmoid} to the $\Delta z$ histogram:
                    \[f(\Delta z) = \frac{a}{1+e^{b(\sigma-|\Delta z|)}} + c\]
                \item $\sigma_{\text{vtx-vtx}}$ = inflection half-width $=$ \textbf{smallest resolvable separation}
            \end{itemize}
        \end{column}
        \begin{column}{0.62\linewidth}

\begin{frame}[shrink=15]{Matching \& Vertex Categories (greedy closest-first)}
    \begin{columns}[c]
        \begin{column}{0.42\linewidth}
            \footnotesize
            We match reco to truth \textbf{by $z$-position} within $\pm\sigma_{\text{vtx-vtx}}$,
            greedily locking in the closest pairs 1-to-1, then label:
            \begin{itemize}\setlength\itemsep{0.35em}
                \item \textcolor{mygreen!70!black}{\textbf{Clean}}: reco keeps its own dedicated truth
                \item \textcolor{myorange!85!black}{\textbf{Merged}}: a second truth in the window is absorbed
                \item \textcolor{myblue}{\textbf{Split}}: extra reco for a truth already claimed
                \item \textcolor{myred}{\textbf{Fake}}: reco with no truth in its window
            \end{itemize}
            \vspace{0.2em}
            \[\varepsilon = \frac{N_\text{clean}+N_\text{merged}}{N_\text{truth}}\quad(\ntrk\geq2)\]
            \vspace{-0.6em}

            {\scriptsize\textcolor{pennred}{\textbf{Convention:} a matched reco is
            credited with \emph{every} unmatched truth in its window, so one vertex can
            be credited with several.} Close to the ATLAS ``merged'' definition, and
            worth \textbf{$+9.7$ to $+15.2$ points} over a strict one-to-one count.
            Applied to AMVF and PV-Finder alike throughout.}
        \end{column}

\begin{frame}[shrink=15]{The Two Closure Identities, and What Divides by What}
    \small
    \textbf{The algorithm}, \texttt{efficiency\_res\_optimized\_atlas.py:258--311}:
    candidate pairs are every reco $i$ with every truth $j$ where
    $|z_j - z_i| \leq$ window; those pairs are assigned greedily, closest first,
    one-to-one. Then:

    \vspace{0.3em}
    \begin{center}\footnotesize
    \begin{tabular}{ll}
        \hline
        \textbf{label} & \textbf{condition} \\
        \hline
        \textcolor{mygreen!70!black}{\textbf{clean}}  & won a dedicated truth, nothing unclaimed left in its window \\
        \textcolor{myorange!85!black}{\textbf{merged}} & won a truth \emph{and} absorbed $\geq 1$ extra unclaimed truth \\
        \textcolor{myblue}{\textbf{split}}   & won nothing, but a truth \emph{is} in its window --- a closer reco took it \\
        \textcolor{myred}{\textbf{fake}}    & won nothing and \emph{no} truth is in its window \\
        \hline
    \end{tabular}
    \end{center}
    \vspace{0.2em}

    Truth side: won a dedicated reco $\to$ \textbf{clean} (tc); absorbed by a merged
    reco $\to$ \textbf{merged} (tm); otherwise \textbf{missed}.

    \vspace{0.4em}
    \begin{center}
    \colorbox{black!5}{\parbox{0.93\linewidth}{\footnotesize\ttfamily
    reco-side:\ \ clean + merged + split + fake\ \ \ \ \ \ \ \ \ = n\_reco\ \ \ (101.01/evt)\\
    truth-side:\ truth\_clean + truth\_merged + truth\_missed = n\_truth\ \ (111.32/evt)}}
    \end{center}
    \vspace{0.3em}

    \textcolor{pennred}{\textbf{Efficiency $=(\text{tc}+\text{tm})/n_\text{truth}$ is
    truth-side. Fake rate $=$ fake$/N_\text{events}$ is reco-side.}} They divide by
    different things and \textbf{do not sum to anything} --- there is no
    ``efficiency $+$ fake $= 1$''.
\end{frame}

\begin{frame}{The Cost Model}
    \begin{columns}[t]
        \begin{column}{0.5\linewidth}
            \small
            \textbf{Why the cost is flat, by construction}
            \begin{itemize}
                \setlength\itemsep{0.3em}
                \item The network maps a track set onto a \textbf{fixed 12{,}000-bin
                    grid}. The grid does not grow with $\mu$, and there is no seeding,
                    no per-vertex fit and no iteration to convergence
                \item What does grow with $\mu$ is the track count, and it enters only
                    through the set-invariant MLP
                \item AMVF instead pays a \textbf{combinatorial cost that grows with the
                    vertex count}, exactly the quantity HL-LHC is designed to increase
            \end{itemize}
        \end{column}
        \begin{column}{0.5\linewidth}
            \small
            \textbf{What is actually measured in this repository}
            \begin{itemize}
                \setlength\itemsep{0.3em}
                \item Full \textbf{finder\,+\,GNN chain $\sim$11\,ms/event} after
                    optimization
                \item \textbf{Peak finding $61 \to 0.07$\,ms} (JIT, bit-exact against
                    the reference implementation)
                \item \textbf{GNN forward 3.8\,ms}, never the bottleneck
            \end{itemize}

            benchmark under the HEPScore protocol has \textbf{no producing code in this

\begin{frame}{Choosing a Matching Window That Favours Neither}
    \begin{columns}[t]
        \begin{column}{0.5\linewidth}
            \small
            \textbf{$\sigma_{\text{vtx-vtx}}$ is the wrong quantity to build a window from.}
            It measures how far apart two vertices must be to be resolved
            \emph{separately}. ``Did the algorithm find this vertex'' needs the
            \textbf{core position resolution}.
            \vspace{0.3em}
            \centering
            \footnotesize
            \begin{tabular}{lccc}
                \toprule
                & PV-Finder & AMVF & ratio \\
                \midrule
                $\sigma_{\text{vtx-vtx}}$ & $0.218$\,mm & $0.305$\,mm & \textcolor{myred}{$\mathbf{1.40}$} \\
                core position res. & $55.1\,\mu$m & $56.5\,\mu$m & \textcolor{mygreen!70!black}{$\mathbf{1.03}$} \\
                \bottomrule
            \end{tabular}
            \vspace{0.4em}
            \raggedright

            A window built on $\sigma_{\text{vtx-vtx}}$ is unfair \emph{by
            construction}; one built on the core resolution is fair by construction.
            Hence \(\mathbf{3\times 56.5\,\mu m = 0.17}\)\,\textbf{mm}, three times the
            worse of the two, fixed before any comparison was read.
        \end{column}
        \begin{column}{0.5\linewidth}
            \small
            \textbf{And a round $0.5$\,mm is measurably too loose.} At PU200 the local
            truth density is ${\sim}0.9$ vertices/mm, so a reco vertex lands near
            \emph{some} truth vertex whether or not it found it. Displace the whole
            reco list by $3$\,mm and re-match: whatever efficiency survives is pure
            coincidence.
            \vspace{0.3em}
            \centering
            \footnotesize
            \begin{tabular}{lcccc}
                \toprule
                common window & $0.10$ & $\mathbf{0.17}$ & $\mathbf{0.50}$ & $1.00$\,mm \\
                \midrule
                accidental eff. & $11.1$ & $\mathbf{18.6}$ & $\mathbf{45.4}$ & $68.1$\,\% \\
                \bottomrule
            \end{tabular}
            \vspace{0.4em}
            \raggedright

            \textbf{At $0.5$\,mm nearly half the measured efficiency is coincidence};
            at $0.17$\,mm it is under a fifth. The floor is the same for both
            algorithms to within $0.3$ points, so it does not bias the
            \emph{comparison} --- but it destroys the meaning of any \emph{absolute}
            number quoted at a loose window.
        \end{column}
    \end{columns}
\end{frame}

\partdivider{IV}{Track-to-vertex association (TTVA)}

\begin{frame}{TTVA: What It Is and Why We Need It}
    \textbf{Vertex finding tells us \emph{where} the vertices are, but not \emph{which tracks belong to which}.}
    \vspace{0.5em}
    \begin{center}
        tracks \(\to\) PV-Finder histogram \(\to\) peak finding \(\to\) candidate vertices
        \(\to\) \textcolor{pennred}{\textbf{TTVA}} \(\to\) per-vertex track lists
    \end{center}
    \vspace{0.5em}
    \begin{columns}[t]
        \begin{column}{0.5\linewidth}
            \textbf{Why analyses need track lists:}
            \begin{itemize}
                \setlength\itemsep{0.35em}
                \item Hard-scatter choice $=$ vertex with largest $\sum p_T^2$ of \emph{its} tracks
                \item Pile-up jet rejection, and any per-vertex quantity (mass, multiplicity, pointing)
            \end{itemize}
        \end{column}
        \begin{column}{0.5\linewidth}
            \textbf{Why we need it for a fair comparison:}
            \begin{itemize}
                \setlength\itemsep{0.35em}
                \item AMVF gets track lists free from its fit; PV-Finder does not
                \item With TTVA both chains are judged by the \emph{same} track-based purity criteria
                \item Design choice: \textbf{train on MC truth vertices}, \textbf{deploy on PV-Finder peaks}
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{TTVA: The Event as a Graph}
    \begin{columns}[c]
        \begin{column}{0.52\linewidth}

\begin{frame}{TTVA: A Small Graph Attention Network}
    \begin{columns}[c]
        \begin{column}{0.55\linewidth}
            \begin{enumerate}
                \setlength\itemsep{0.4em}
                \item Embed track (8) and vertex (2) features into 32 dimensions
                \item \textbf{2 rounds of heterogeneous graph attention} (4 heads): each node updates from its neighbours, the \textbf{edge attributes steer the attention}
                \item[] {\small\(\Rightarrow\) a vertex that already attracts compatible tracks attracts the next one more strongly, context a per-pair cut cannot express}
                \item A 3-layer MLP on each (track, vertex) embedding \(\to\) \textbf{one score per edge}
            \end{enumerate}
            \vspace{0.2em}
            {\footnotesize\textbf{Size \& training:} \textbf{39{,}521 parameters} ($90\times$
            smaller than PV-Finder), $3.8$\,ms/event; BCE loss, Adam, trained on 180k PU200 events in 3.2\,h.}
        \end{column}
        \begin{column}{0.45\linewidth}
            \centering
            \includegraphics[width=0.98\linewidth, keepaspectratio]{images/edge_score_distributions.png}\\[2pt]
            {\scriptsize\textcolor{black!55}{True vs false edges ($\langle\mu\rangle\!\approx\!60$):
            the classifier is nearly saturated, with ROC \textbf{AUC 0.998} on $45$M edges.}}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{TTVA: From Scores to Vertices}
    \small

            but only if that score clears a threshold \(t\); otherwise it stays unassigned.
            The model runs \textbf{once}, and \(t\) is a post-hoc knob (scores saturate near 1,

                \item \textcolor{myorange!85!black}{\textbf{Merged}}: no such majority

\begin{frame}[shrink=15]{Conclusions}
    \begin{columns}[t]
        \begin{column}{0.5\linewidth}
            \textbf{What we've shown}
            \begin{enumerate}
                \setlength\itemsep{0.4em}
                \item End-to-end trainable model with the KDE removed from training; validated on Run 2 \& Run 3 data
                \item At low pileup, matches AMVF and \textbf{beats it in resolution} ($0.42$ vs $0.79$\,mm)
                \item At \textbf{PU200}, on genuinely held-out events and at a window
                    that favours neither algorithm, PV-Finder leads AMVF by
                    \textbf{$+3.4 \pm 0.1$ efficiency points} and
                    $0.60 \pm 0.09$ fewer surplus vertices/event,
                    with $\sigma_{\text{vtx-vtx}} = 0.229$\,mm against $0.305$
                \item \textbf{TTVA completes the chain} at $\langle\mu\rangle\!\approx\!60$
                    and costs \(\sim\)$11$\,ms/event. At \textbf{PU200} it reaches
                    \textbf{94.0\,\%} of the oracle ($0.6843$ clean/truth against
                    AMVF's $0.3094$), so the chain is \textbf{finder-limited}
            \end{enumerate}
        \end{column}
        \begin{column}{0.5\linewidth}
            \textbf{Three framing findings}
            \begin{itemize}
                \setlength\itemsep{0.35em}
                \item Capacity is \textbf{not} the bottleneck: a 5$\times$ model and an LR/optimizer/loss sweep all plateau together
                \item Part of the residual inefficiency is \textbf{physical}: $\sim$17\% of adjacent PU200 vertices are closer than the resolution
                \item \textbf{Post-hoc fake suppression is done}: 0 of 817 configurations beat the deployed operating point, so the remaining gains have to come from training
                \item The headroom is the \textbf{finder's}: $21.9$ of the $27.2$ points between truth and the association oracle are truth vertices with no peak
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[shrink=15]{Next Steps}
    \begin{columns}[t]
        \begin{column}{0.5\linewidth}
            \textbf{Finder side --- all of it needs a retrain}

                    upsample. Surplus peaks carry a $35\%$ mod-4 modulation that

                    a finer grid in the peak region
                \item \textbf{In-model objectness head} or a fake-aware loss: the only
                    item that attacks surplus peaks at the source
            \end{itemize}
